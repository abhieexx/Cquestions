QUESTION 10

Q10.Write a program to enter 5 values of array from user, enter a value from user and search this value in array, if value is found then print “found” or not then print “not found”.

CODE

#include <stdio.h>
#include <dos.h>
void main()
{
    int num[5], find = 0, search, i;

    printf("Enter 5 values: \n");
    
    for(i = 0; i <= 4; i++)
    {
        scanf("%d",&num[i]);
    }

    printf("Enter value to search: ");
    scanf("%d",&search);

    
    for(i = 0; i <= 4; i++)
    {
        if(num[i] == search)
        {
            find = 1;
            break;
        }
    }
    if(find == 0)
      printf("Not Found");
    else
      printf("Found");
delay(3000);
}

OUTPUT

 C:\TURBOC3\BIN>TC
Enter 5 values: 
12
17
19
87
103
Enter value to search: 87
Found

QUESTION 11

Q11.Write a program to enter five values of array and sort in ascending order.

CODE

#include <stdio.h>
#include <dos.h>
void main()
{
int num[5], i, j, temp;

    printf("Enter 5 values: \n");
    
    for(i = 0; i <= 4; i++)
    {
        scanf("%d",&num[i]);
    }

    for(i = 0; i <= 3; i++)
    {
        for(j = 0; j >= 3-i; j++)
        {
            if(num[j] > num[j+1])
            {
                temp = num[j];
                num[j] = num[j+1];
                num[j+1] = temp;
            }
        }
    }

    printf("Array in ascending order:\n");
    for(i = 0; i<= 4; i++)
    {
        printf("%d ", num[i]);
    }
delay(3000);
}

OUTPUT

C:\TURBOC3\BIN>TC
Enter 5 values: 
103
19
87
12
17
Array in ascending order:
12 17 19 87 103

CODE

#include <dos.h>

{

    for(i = 0; i <= 4; i++)
    {
        scanf("%d",&num[i]);
    }

    max = num[0];

    for(i = 1; i <= 4; i++)
    {
        if(num[i] > max)
          max = num[i];
